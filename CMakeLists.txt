cmake_minimum_required(VERSION 3.24)
set(SNOW_VERSION_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/src/version.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SNOW_VERSION_HEADER}")
file(STRINGS "${SNOW_VERSION_HEADER}" SNOW_VERSION_LINE
    REGEX "^#define SPEEECIAL_VERSION \"[0-9]+\\.[0-9]+\\.[0-9]+\"$")
if(NOT SNOW_VERSION_LINE MATCHES "^#define SPEEECIAL_VERSION \"([0-9]+\\.[0-9]+\\.[0-9]+)\"$")
    message(FATAL_ERROR "Cannot read application version from src/version.hpp")
endif()
set(SNOW_VERSION "${CMAKE_MATCH_1}")
project(speeecial VERSION "${SNOW_VERSION}" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(snow_update_core STATIC src/update_core.cpp)
target_include_directories(snow_update_core PUBLIC src)
if(MSVC)
    target_compile_options(snow_update_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(snow_update_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(snow_core STATIC src/core.cpp src/scene.cpp)
target_include_directories(snow_core PUBLIC src)
if(MSVC)
    target_compile_options(snow_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(snow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    enable_language(RC)
    set(SNOW_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
    string(REPLACE "." "," SNOW_VERSION_RESOURCE "${PROJECT_VERSION}.0")
    configure_file(resources/speeecial.manifest.in "${SNOW_GENERATED_DIR}/speeecial.manifest" @ONLY)
    set(SNOW_RESOURCE_FILE "${SNOW_GENERATED_DIR}/speeecial.rc")
    configure_file(resources/speeecial.rc.in "${SNOW_RESOURCE_FILE}" @ONLY)
    add_executable(speeecial WIN32
        src/updater.cpp src/main.cpp src/app.cpp src/desktop.cpp src/renderer.cpp src/settings.cpp src/migration.cpp src/tray_menu.cpp
        "${SNOW_RESOURCE_FILE}")
    set_target_properties(speeecial PROPERTIES OUTPUT_NAME "speeecial")
    target_include_directories(speeecial PRIVATE src resources)
    target_compile_definitions(speeecial PRIVATE
        UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
        _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(speeecial PRIVATE snow_core snow_update_core winhttp bcrypt version
        user32 gdi32 shell32 ole32 advapi32 dwmapi shcore wtsapi32 psapi
        d3d11 dxgi d2d1 dcomp uuid comctl32)
    if(MSVC)
        target_compile_options(speeecial PRIVATE /W4 /permissive- /utf-8)
        # The manifest (DPI awareness, asInvoker) is embedded by the .rc file.
        target_link_options(speeecial PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_compile_options(speeecial PRIVATE -Wall -Wextra -Wpedantic)
        target_link_options(speeecial PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS speeecial RUNTIME DESTINATION .)
endif()

option(SPEEECIAL_WINDOWS_UPDATE_TESTS "Build Windows updater process/rollback tests (no network)" OFF)
if(WIN32 AND BUILD_TESTING AND SPEEECIAL_WINDOWS_UPDATE_TESTS)
    add_executable(snow_update_old_fixture tests/windows_updater_fixture.cpp "${SNOW_RESOURCE_FILE}")
    add_executable(snow_update_new_fixture tests/windows_updater_fixture.cpp "${SNOW_RESOURCE_FILE}")
    target_compile_definitions(snow_update_new_fixture PRIVATE SNOW_UPDATE_NEW_FIXTURE)
    add_executable(snow_windows_update_tests tests/windows_updater_tests.cpp)
    get_target_property(snow_update_definitions speeecial COMPILE_DEFINITIONS)
    foreach(target snow_update_old_fixture snow_update_new_fixture snow_windows_update_tests)
        target_include_directories(${target} PRIVATE src resources)
        target_compile_definitions(${target} PRIVATE ${snow_update_definitions})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        elseif(MINGW)
            target_link_options(${target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
        endif()
    endforeach()
    if(MSVC)
        target_link_options(snow_update_old_fixture PRIVATE /MANIFEST:NO)
        target_link_options(snow_update_new_fixture PRIVATE /MANIFEST:NO)
    endif()
    target_link_libraries(snow_windows_update_tests PRIVATE snow_update_core winhttp bcrypt version ole32 shell32 user32)
    add_test(NAME snow_windows_update_tests COMMAND snow_windows_update_tests
        $<TARGET_FILE:snow_update_old_fixture> $<TARGET_FILE:snow_update_new_fixture>)
    set_tests_properties(snow_windows_update_tests PROPERTIES LABELS "windows;updater" RUN_SERIAL TRUE TIMEOUT 240)
endif()
