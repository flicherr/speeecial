cmake_minimum_required(VERSION 3.24)
project(snowdesk VERSION 1.0.10 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(snow_core STATIC src/core.cpp src/scene.cpp)
target_include_directories(snow_core PUBLIC src)
if(MSVC)
    target_compile_options(snow_core PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(snow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    enable_language(RC)
    add_executable(snowdesk WIN32
        src/main.cpp src/app.cpp src/desktop.cpp src/renderer.cpp src/settings.cpp src/tray_menu.cpp
        resources/snowdesk.rc)
    set_target_properties(snowdesk PROPERTIES OUTPUT_NAME "speeecial")
    target_include_directories(snowdesk PRIVATE src resources)
    target_compile_definitions(snowdesk PRIVATE
        UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
        _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(snowdesk PRIVATE snow_core
        user32 gdi32 shell32 ole32 advapi32 dwmapi shcore wtsapi32 psapi
        d3d11 dxgi d2d1 dcomp uuid comctl32)
    if(MSVC)
        target_compile_options(snowdesk PRIVATE /W4 /permissive- /utf-8)
        # The manifest (DPI awareness, asInvoker) is embedded by the .rc file.
        target_link_options(snowdesk PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_compile_options(snowdesk PRIVATE -Wall -Wextra -Wpedantic)
        target_link_options(snowdesk PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS snowdesk RUNTIME DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(snow_core_tests tests/core_tests.cpp)
    target_link_libraries(snow_core_tests PRIVATE snow_core)
    add_test(NAME snow_core_tests COMMAND snow_core_tests)
endif()

option(SNOWDESK_WINDOWS_GRAPHICS_TESTS "Build opt-in live Windows/GPU rebuild stress test" OFF)
if(WIN32 AND BUILD_TESTING AND SNOWDESK_WINDOWS_GRAPHICS_TESTS)
    add_executable(snow_windows_graphics_tests
        tests/windows_graphics_tests.cpp src/desktop.cpp src/renderer.cpp)
    target_include_directories(snow_windows_graphics_tests PRIVATE src resources)
    get_target_property(snow_win_definitions snowdesk COMPILE_DEFINITIONS)
    get_target_property(snow_win_libraries snowdesk LINK_LIBRARIES)
    target_compile_definitions(snow_windows_graphics_tests PRIVATE ${snow_win_definitions})
    target_link_libraries(snow_windows_graphics_tests PRIVATE ${snow_win_libraries})
    if(MSVC)
        target_compile_options(snow_windows_graphics_tests PRIVATE /W4 /permissive- /utf-8)
    elseif(MINGW)
        target_link_options(snow_windows_graphics_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME snow_windows_graphics_tests COMMAND snow_windows_graphics_tests)
    set_tests_properties(snow_windows_graphics_tests PROPERTIES LABELS "windows;graphics;interactive" RUN_SERIAL TRUE)
endif()

option(SNOWDESK_WINDOWS_MENU_TESTS "Build opt-in native tray menu resource/callback tests" OFF)
if(WIN32 AND BUILD_TESTING AND SNOWDESK_WINDOWS_MENU_TESTS)
    add_executable(snow_windows_menu_tests tests/windows_menu_tests.cpp src/tray_menu.cpp resources/snowdesk.rc)
    target_include_directories(snow_windows_menu_tests PRIVATE src resources)
    get_target_property(snow_menu_definitions snowdesk COMPILE_DEFINITIONS)
    target_compile_definitions(snow_windows_menu_tests PRIVATE ${snow_menu_definitions})
    target_link_libraries(snow_windows_menu_tests PRIVATE user32 gdi32 advapi32 shcore comctl32)
    if(MSVC)
        target_compile_options(snow_windows_menu_tests PRIVATE /W4 /permissive- /utf-8)
        target_link_options(snow_windows_menu_tests PRIVATE /MANIFEST:NO)
    elseif(MINGW)
        target_link_options(snow_windows_menu_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME snow_windows_menu_tests COMMAND snow_windows_menu_tests)
    set_tests_properties(snow_windows_menu_tests PROPERTIES LABELS "windows;menu" RUN_SERIAL TRUE)
endif()
